    <!-- ======================================================================
         PAGE SCRIPT (dashboard.html)
         Deliberately page-local and deliberately dumb: it holds mock rows and
         draws the table and the KPI tiles. In Milestone 7 the two constants
         below are replaced by:

             const response = await fetch("/api/engagements", { credentials: "same-origin" });
             const MOCK_ENGAGEMENTS = await response.json();

         and nothing else in this file has to change.
         ====================================================================== -->
    <script>
      /* Status values mirror the future `engagements.status` column. */
      var MOCK_ENGAGEMENTS = [
        {
          reference: "ORL-482913",
          service: "ORACLE AI",
          status: "active",
          slot: "2026-09-28T09:00:00",
          findings: 7,
          invoice_state: "outstanding",
          invoice_amount: 15000
        },
        {
          reference: "ORL-517204",
          service: "PHANTASM",
          status: "scoping",
          slot: "2026-10-12T08:00:00",
          findings: 0,
          invoice_state: "not_issued",
          invoice_amount: 22000
        },
        {
          reference: "ORL-533187",
          service: "CHAINBREAK",
          status: "pending",
          slot: "2026-11-02T09:00:00",
          findings: 0,
          invoice_state: "not_issued",
          invoice_amount: 18000
        },
        {
          reference: "ORL-401662",
          service: "ORACLE AI",
          status: "completed",
          slot: "2026-06-15T09:00:00",
          findings: 12,
          invoice_state: "paid",
          invoice_amount: 15000
        },
        {
          reference: "ORL-388041",
          service: "PHANTASM",
          status: "cancelled",
          slot: "2026-05-20T09:00:00",
          findings: 0,
          invoice_state: "cancelled",
          invoice_amount: 0
        }
      ];

      /* --- tiny local helpers (this script runs before js/main.js loads, so it
             does not rely on the shared API in window.ORL) ----------------- */
      var dashboardStatus = document.getElementById("dashboard-status");

      function announce(message) {
        if (dashboardStatus) {
          dashboardStatus.textContent = message;
        }
      }

      function el(tagName, className, text) {
        var node = document.createElement(tagName);
        if (className) {
          node.className = className;
        }
        if (typeof text === "string") {
          node.textContent = text;
        }
        return node;
      }

      function formatSlot(iso) {
        var date = new Date(iso);
        if (isNaN(date.getTime())) {
          return iso;
        }
        return date.toLocaleString("en-US", {
          year: "numeric", month: "short", day: "numeric",
          hour: "2-digit", minute: "2-digit"
        });
      }

      function formatSlotShort(iso) {
        var date = new Date(iso);
        return isNaN(date.getTime())
          ? iso
          : date.toLocaleDateString("en-US", { month: "short", day: "numeric" });
      }

      /* --- KPI tiles -------------------------------------------------- */
      function renderStats() {
        var container = document.getElementById("dashboard-stats");
        if (!container) {
          return;
        }

        var openStatuses = ["pending", "scoping", "active"];
        var open = MOCK_ENGAGEMENTS.filter(function (engagement) {
          return openStatuses.indexOf(engagement.status) !== -1;
        });
        var findings = MOCK_ENGAGEMENTS.reduce(function (total, engagement) {
          return total + engagement.findings;
        }, 0);
        var upcoming = open
          .filter(function (engagement) {
            return new Date(engagement.slot).getTime() >= Date.now();
          })
          .sort(function (a, b) {
            return new Date(a.slot) - new Date(b.slot);
          })[0];
        var outstanding = MOCK_ENGAGEMENTS.filter(function (engagement) {
          return engagement.invoice_state === "outstanding";
        }).reduce(function (total, engagement) {
          return total + engagement.invoice_amount;
        }, 0);

        var tiles = [
          { label: "Open engagements", value: String(open.length) },
          { label: "Findings logged", value: String(findings) },
          {
            label: "Next start window",
            value: upcoming ? formatSlotShort(upcoming.slot) : "None scheduled"
          },
          {
            label: "Outstanding invoices",
            value: outstanding ? "$" + outstanding.toLocaleString("en-US") : "$0"
          }
        ];

        container.replaceChildren();
        tiles.forEach(function (tile) {
          var card = el("div", "card card--stat");
          card.appendChild(el("span", "card__label", tile.label));
          card.appendChild(el("p", "card__value", tile.value));
          container.appendChild(card);
        });
      }

      /* --- table rows ------------------------------------------------- */
      function buildRow(engagement) {
        var row = document.createElement("tr");

        var reference = document.createElement("th");
        reference.scope = "row";
        reference.className = "table__row-head";
        reference.textContent = engagement.reference;
        row.appendChild(reference);

        /* Every td carries data-label so the stacked mobile view can print the
           column heading (see the <=600px rules in css/responsive.css). */
        var service = document.createElement("td");
        service.setAttribute("data-label", "Service");
        service.textContent = engagement.service;
        row.appendChild(service);

        var status = document.createElement("td");
        status.setAttribute("data-label", "Status");
        var badge = el("span", "badge badge--" + engagement.status, engagement.status);
        status.appendChild(badge);
        row.appendChild(status);

        var slot = document.createElement("td");
        slot.setAttribute("data-label", "Requested slot");
        slot.textContent = formatSlot(engagement.slot);
        row.appendChild(slot);

        var findings = document.createElement("td");
        findings.setAttribute("data-label", "Findings");
        findings.textContent = String(engagement.findings);
        row.appendChild(findings);

        /* Actions. Cancel is disabled for anything already completed or
           cancelled - the only rule these stubs enforce. */
        var isCancellable =
          engagement.status !== "completed" && engagement.status !== "cancelled";

        var actions = document.createElement("td");
        actions.setAttribute("data-label", "Actions");
        var group = el("div", "table__actions");

        var viewButton = el("button", "btn btn--ghost btn--sm", "View");
        viewButton.type = "button";
        viewButton.addEventListener("click", function () {
          announce(
            "View " + engagement.reference +
              ": the findings view is a stub in this milestone."
          );
        });
        group.appendChild(viewButton);

        var cancelButton = el("button", "btn btn--danger btn--sm", "Cancel");
        cancelButton.type = "button";
        cancelButton.disabled = !isCancellable;
        if (!isCancellable) {
          cancelButton.title =
            "Completed and cancelled engagements cannot be cancelled.";
        }
        cancelButton.addEventListener("click", function () {
          announce(
            "Cancel " + engagement.reference +
              ": this would request cancellation from your engagement lead."
          );
        });
        group.appendChild(cancelButton);

        actions.appendChild(group);
        row.appendChild(actions);

        return row;
      }

      function renderTable() {
        var tbody = document.querySelector("#engagements-table tbody");
        if (!tbody) {
          return;
        }
        tbody.replaceChildren();
        MOCK_ENGAGEMENTS.forEach(function (engagement) {
          tbody.appendChild(buildRow(engagement));
        });
      }

      /* Draw the page. */
      renderStats();
      renderTable();
      announce(MOCK_ENGAGEMENTS.length + " engagements loaded from mock data.");
    </script>

    <!-- ======================================================================
         PAGE SCRIPT (admin.html)
         Wrapped in DOMContentLoaded so the shared scripts - including
         js/data/services.js and js/data/resources.js - have run before we read
         those globals. Milestone 7 replaces the two mock arrays with fetch()
         calls and swaps the stub buttons for real CRUD handlers.
         ====================================================================== -->
    <script>
      document.addEventListener("DOMContentLoaded", function () {
        /* Engagement statuses allowed by the future `engagements.status` column. */
        var STATUSES = ["pending", "scoping", "active", "completed", "cancelled"];

        /* Mock engagements and inquiries (these live in a database later). */
        var MOCK_ENGAGEMENTS_ADMIN = [
          { reference: "ORL-482913", client: "Northwind Logistics", service: "ORACLE AI", slot: "2026-09-28T09:00:00", status: "active" },
          { reference: "ORL-517204", client: "Helios Bioworks", service: "PHANTASM", slot: "2026-10-12T08:00:00", status: "scoping" },
          { reference: "ORL-533187", client: "Verge Cloud", service: "CHAINBREAK", slot: "2026-11-02T09:00:00", status: "pending" },
          { reference: "ORL-401662", client: "Northwind Logistics", service: "ORACLE AI", slot: "2026-06-15T09:00:00", status: "completed" }
        ];

        var MOCK_INQUIRIES = [
          { received: "2026-09-22T14:05:00", contact: "J. Ellis", company: "Northwind Logistics", message: "External assessment of the customer portal before our Q4 audit window." },
          { received: "2026-09-21T09:41:00", contact: "R. Okonkwo", company: "Helios Bioworks", message: "Tabletop plus assumed-breach rehearsal across four clinical sites." },
          { received: "2026-09-19T16:17:00", contact: "M. Lindqvist", company: "Verge Cloud", message: "CI/CD provenance review after a dependency scare." }
        ];

        /* --- helpers ------------------------------------------------------ */
        var statusLine = document.getElementById("admin-status");

        function announce(message) {
          if (statusLine) {
            statusLine.textContent = message;
          }
        }

        function el(tagName, className, text) {
          var node = document.createElement(tagName);
          if (className) {
            node.className = className;
          }
          if (typeof text === "string") {
            node.textContent = text;
          }
          return node;
        }

        /** A td carrying the data-label used by the stacked mobile layout. */
        function cell(label, content) {
          var td = document.createElement("td");
          td.setAttribute("data-label", label);
          if (content instanceof Node) {
            td.appendChild(content);
          } else {
            td.textContent = content;
          }
          return td;
        }

        /** Row header cell: the first column of every table here. */
        function rowHead(text) {
          var th = document.createElement("th");
          th.scope = "row";
          th.className = "table__row-head";
          th.textContent = text;
          return th;
        }

        /** Stub action buttons: they announce intent instead of acting. */
        function actionCell(recordName, labels) {
          var group = el("div", "table__actions");
          labels.forEach(function (label) {
            var isDestructive = label === "Delete";
            var button = el(
              "button",
              "btn btn--sm " + (isDestructive ? "btn--danger" : "btn--ghost"),
              label
            );
            button.type = "button";
            button.addEventListener("click", function () {
              announce(
                label + " " + recordName +
                  ": this control is a labelled stub until the back end exists."
              );
            });
            group.appendChild(button);
          });
          return cell("Actions", group);
        }

        function formatSlot(iso) {
          var date = new Date(iso);
          return isNaN(date.getTime())
            ? iso
            : date.toLocaleString("en-US", {
                year: "numeric", month: "short", day: "numeric",
                hour: "2-digit", minute: "2-digit"
              });
        }

        function formatPrice(value) {
          return "$" + Number(value).toLocaleString("en-US", {
            maximumFractionDigits: 0
          });
        }

        /** Replace the rows of one table body. */
        function fill(tbodySelector, rows) {
          var tbody = document.querySelector(tbodySelector);
          if (!tbody) {
            return;
          }
          tbody.replaceChildren();
          rows.forEach(function (row) {
            tbody.appendChild(row);
          });
        }

        /* --- services table (from js/data/services.js) -------------------- */
        function renderServices() {
          var rows = (typeof services === "undefined" ? [] : services).map(
            function (service) {
              var row = document.createElement("tr");
              row.appendChild(rowHead(String(service.service_id)));
              row.appendChild(cell("Name", service.name));
              row.appendChild(cell("Category", service.category));
              row.appendChild(cell("Delivery method", service.delivery_method));
              row.appendChild(cell("Base price", formatPrice(service.base_price)));
              row.appendChild(actionCell(service.name, ["Edit", "Delete"]));
              return row;
            }
          );
          fill("#admin-services-table tbody", rows);
          return rows.length;
        }

        /* --- engagement register with the status dropdown ---------------- */
        function renderEngagements() {
          var rows = MOCK_ENGAGEMENTS_ADMIN.map(function (engagement) {
            var row = document.createElement("tr");
            row.appendChild(rowHead(engagement.reference));
            row.appendChild(cell("Client", engagement.client));
            row.appendChild(cell("Service", engagement.service));
            row.appendChild(cell("Requested slot", formatSlot(engagement.slot)));

            /* A <select> inside a table cell has no visible label of its own,
               so it gets an aria-label that names the row it belongs to. */
            var select = document.createElement("select");
            select.className = "select select--inline";
            select.setAttribute(
              "aria-label",
              "Status for engagement " + engagement.reference
            );
            STATUSES.forEach(function (status) {
              var option = document.createElement("option");
              option.value = status;
              option.textContent = status;
              if (status === engagement.status) {
                option.selected = true;
              }
              select.appendChild(option);
            });
            select.addEventListener("change", function () {
              announce(
                "Status for " + engagement.reference + " set to " + select.value +
                  " - not saved, there is no back end yet."
              );
            });
            row.appendChild(cell("Status", select));

            row.appendChild(actionCell(engagement.reference, ["View", "Assign"]));
            return row;
          });
          fill("#admin-engagements-table tbody", rows);
          return rows.length;
        }

        /* --- inquiry queue ----------------------------------------------- */
        function renderInquiries() {
          var rows = MOCK_INQUIRIES.map(function (inquiry) {
            var row = document.createElement("tr");
            row.appendChild(rowHead(formatSlot(inquiry.received)));
            row.appendChild(cell("Contact", inquiry.contact));
            row.appendChild(cell("Company", inquiry.company));
            row.appendChild(cell("Message", inquiry.message));
            row.appendChild(actionCell(inquiry.company, ["Review", "Delete"]));
            return row;
          });
          fill("#admin-inquiries-table tbody", rows);
          return rows.length;
        }

        /* --- vault library (from js/data/resources.js) -------------------- */
        function renderVault() {
          var rows = (typeof resources === "undefined" ? [] : resources).map(
            function (resource) {
              var row = document.createElement("tr");
              row.appendChild(rowHead(String(resource.resource_id)));
              row.appendChild(cell("Title", resource.title));
              row.appendChild(cell("Category", resource.category));
              row.appendChild(cell("Published", resource.published_at));
              row.appendChild(actionCell(resource.title, ["Edit", "Delete"]));
              return row;
            }
          );
          fill("#admin-vault-table tbody", rows);
          return rows.length;
        }

        /* --- draw the console ------------------------------------------- */
        var serviceCount = renderServices();
        var engagementCount = renderEngagements();
        var inquiryCount = renderInquiries();
        var vaultCount = renderVault();

        announce(
          serviceCount + " services, " + engagementCount + " engagements, " +
            inquiryCount + " inquiries and " + vaultCount +
            " vault entries loaded from static mock data."
        );
      });
    </script>
